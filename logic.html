CREATE aligned_create — 建表 / 扩展列组
aligned_create(table, group, columns, root=..., partition_template=...) — 建表时创建 index 列组 + 默认分区占位 parquet（0 行，footer 携带 schema）。 扩展列组时为每个已有分区写 N 行全 NULL 占位 parquet，满足分区对齐契约。分区模板默认 month=%Y-%m。

READ aligned_scan — 多列组并行扫描
SELECT date, symbol, close, alpha001, ma20 FROM aligned_scan('cnstk_ixday') WHERE date = '2026-09-01' — 只打开被选中的列组，Parquet 内只读被选中的列。多个 Parquet Reader 的 Vector 直接组装进同一个 DataChunk，不做 JOIN。 分区裁剪跳过不匹配的分区，Row Group 裁剪跳过不匹配的 RG。

UPDATE INSERT / UPDATE — 按 (symbol, date) 主键写入
INSERT INTO al.cnstk_ixday VALUES ('600000', DATE '2026-10-15', 12.5, ...) — 按 (symbol, date) 主键定位：新行追加到对应分区的末 part（若末 part < 1M 行则原地追加重写，否则新建 part）； 已有行则原地更新。只重写受影响的 part，其他 part 不动。两阶段提交：先写 _tmp/ 再原子 move。

DELETE aligned_drop / DELETE — 删除列组或行
aligned_drop(table, 'factor/alpha') 删除单个列组目录树；aligned_drop(table, 'index') 删除整张表。 标准 DELETE FROM al.cnstk_ixday WHERE date = '2026-07-01' 按主键删除行：删空分区 → 整分区移除，删空中间 part → 原地重写为 0 行空文件保持索引连续。
